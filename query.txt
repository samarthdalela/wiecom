-- Conference Registration Database Schema
CREATE DATABASE upwiecon2026_registration;
USE upwiecon2026_registration;

-- Main registration table
CREATE TABLE tblUPWIECON2026 (
    iRegId INT PRIMARY KEY AUTO_INCREMENT,
    sEventName VARCHAR(50) DEFAULT 'UPWIECON2026',
    sCampus VARCHAR(100) DEFAULT 'Uttarakhand',
    sNielit VARCHAR(10),
    sIEEEMember VARCHAR(20),
    sNationality VARCHAR(20),
    sEarlyBird VARCHAR(20),
    sCategory VARCHAR(50),
    sPaperId VARCHAR(50),
    sPaperTitle VARCHAR(255),
    sName VARCHAR(100) NOT NULL,
    sMobile VARCHAR(15) NOT NULL,
    sEmail VARCHAR(100) NOT NULL,
    sPaperUpload VARCHAR(500),
    iAmount DECIMAL(10,2) NOT NULL,
    sIP VARCHAR(45),
    dtCreated TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    dtUpdated TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    UNIQUE KEY unique_email (sEmail)
);

-- Payment tracking table
CREATE TABLE tblUPWIECON2026Payment (
    iPaymentId INT PRIMARY KEY AUTO_INCREMENT,
    RefId INT NOT NULL,
    EventName VARCHAR(50) DEFAULT 'UPWIECON2026',
    order_id VARCHAR(50) UNIQUE,
    amount DECIMAL(10,2),
    payment_status ENUM('PENDING', 'SUCCESS', 'FAILED', 'CANCELLED') DEFAULT 'PENDING',
    billdesk_txn_id VARCHAR(100),
    billdesk_response TEXT,
    payment_date DATETIME,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (RefId) REFERENCES tblUPWIECON2026(iRegId) ON DELETE CASCADE
);

-- BillDesk configuration
CREATE TABLE billdesk_config (
    id INT PRIMARY KEY AUTO_INCREMENT,
    merchant_id VARCHAR(50) NOT NULL,
    security_id VARCHAR(50) NOT NULL,
    checksum_key VARCHAR(100) NOT NULL,
    return_url VARCHAR(255) NOT NULL,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
